#version 300 es
precision highp float;
precision highp int;
in highp float value;
uniform highp float scale;
out highp vec4 result;
void main ()
{
  mediump float t0 = value;
  result = vec4((value * scale), (t0 * scale), 3.0, 1.0);
}
