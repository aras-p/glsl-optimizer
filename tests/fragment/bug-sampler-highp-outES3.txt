#version 300 es
precision highp float;
precision highp int;
uniform highp sampler2D _CameraDepthTexture;
in highp vec2 varUV;
out mediump vec4 _fragData;
void main ()
{
  mediump vec4 t0 = vec4(varUV, 0.0, 0.0);
  mediump vec4 t1 = textureLod(_CameraDepthTexture, t0.xy, t0.w);
  highp float t2 = t1.x;
  _fragData = vec4(t2);
}

// textures: 1
