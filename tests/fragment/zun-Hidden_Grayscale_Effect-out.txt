uniform float _RampOffset;

{

  vec4 t1 = texture2D(_RampTex, vec2((dot(t0.xyz, vec3(0.22, 0.707, 0.071)) + _RampOffset), 0.5));
  gl_FragData[0] = vec4(t1.xyz, t0.w);
}
