#version 310 es
precision highp float;
precision highp int;
layout(std430) buffer Data { uint value; uint results[4]; };
uniform bool update;
out vec4 color;
void bump()
{
    atomicAdd(value, 1u);
}
void main()
{
    value = 7u;
    uint before = value;
    bump();
    uint after = value;
    if (update) {
        int i = 0;
        atomicExchange(results[i++], 5u);
        atomicCompSwap(results[i++], 5u, 6u);
        results[i] = value;
    }
    color = vec4(float(before), float(after), float(value), 1.0);
}
