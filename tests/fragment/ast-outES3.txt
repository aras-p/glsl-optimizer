#version 300 es
precision highp float;
precision highp int;
out mediump vec4 _fragData;
void main ()
{
  highp int r0;
  highp float r1;
  if ((gl_FragCoord.x == 1.0)) {
    discard;
  }
  r0 = 0;
  r1 = 4.0;
  while (true) {
    r0 = (r0 + 1);
    if ((r0 >= 10)) {
      break;
    }
    r1 = (r1 + 1.0);
  }
  highp float t0 = (4.0 + r1);
  _fragData = vec4(inversesqrt((-1.0 - (t0 * t0))));
}
