#version 300 es
precision highp float;
precision highp int;
out mediump vec4 _fragData;
void main ()
{
  highp int r0;
  highp float r1;
  if ((gl_FragCoord.x == 1.0)) {
    discard;
  }
  r1 = 4.0;
  for (r0 = 0; (r0 < 10); r0++) {
    r1 = (r1 + 1.0);
  }
  highp float t0 = (3.0 + r1);
  _fragData = vec4(inversesqrt((-1.0 - (t0 * t0))));
}
