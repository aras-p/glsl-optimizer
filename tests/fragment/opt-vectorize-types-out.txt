uniform sampler2D maintex;
uniform float factor;
varying vec2 uv;
void main ()
{
  vec4 t0 = texture2D(maintex, uv);
  gl_FragColor = vec4(max(0.123, t0.x), max(0.123, t0.y), min(t0.z, factor), min(t0.w, factor));
}
