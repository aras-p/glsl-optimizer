uniform sampler2D _MainTex;

{
  if ((texture2D(_MainTex, gl_TexCoord[0].xy).w < _Cutoff)) {
    discard;
  }
  vec2 t0 = fract((vec2(1.0, 255.0) * gl_TexCoord[1].w));
  vec2 t1 = ((((gl_TexCoord[1].xy / (gl_TexCoord[1].z + 1.0)) / 1.7777) * 0.5) + 0.5);
  gl_FragData[0] = vec4(t1, (t0.x - (t0.y * 0.00392157)), t0.y);
}
