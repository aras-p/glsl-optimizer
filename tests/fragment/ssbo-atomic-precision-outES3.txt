#version 310 es
precision highp float;
precision highp int;
layout(std430) buffer Data {
  mediump ivec2 values[4];
} data;
uniform mediump int index;
uniform mediump int operand;
out mediump vec4 color;
void main ()
{
  mediump int t0 = atomicAdd(data.values[index][0], operand);
  mediump int t1 = atomicCompSwap(data.values[index][1], t0, operand);
  color = vec4(float(t0), float(t1), 0.0, 1.0);
}
