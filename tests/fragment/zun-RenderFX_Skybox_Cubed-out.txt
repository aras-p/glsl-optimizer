uniform samplerCube _Tex;
uniform vec4 _Tint;
void main ()
{
  vec4 t0 = textureCube(_Tex, gl_TexCoord[0].xyz);
  vec3 t1 = ((t0.xyz - 0.5) + _Tint.xyz);
  gl_FragData[0] = vec4(t1, (t0.w * _Tint.w));
}
