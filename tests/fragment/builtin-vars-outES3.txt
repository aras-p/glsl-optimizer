#version 300 es
precision highp float;
precision highp int;
out mediump vec4 _fragData;
in mediump vec4 uv;
void main ()
{
  mediump vec4 t0 = (uv + gl_FragCoord);
  mediump vec2 t1 = (vec2((t0.x + float(gl_FrontFacing)), t0.y) + gl_PointCoord);
  _fragData = vec4(t1, t0.zw);
}
