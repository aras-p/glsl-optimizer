uniform vec3 uniVal;
void main ()
{
  if ((1.0 < uniVal.x)) {
    gl_FragData[0] = vec4(0.0);
  } else {
    if ((0.001 < uniVal.x)) {
      discard;
    }
    if ((0.001 < uniVal.y)) {
      discard;
    }
    gl_FragData[0] = vec4(1.0);
  }
}
