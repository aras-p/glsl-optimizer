uniform vec4 _Color;
uniform float _Cutoff;
uniform sampler2D _LightBuffer;
uniform sampler2D _MainTex;
uniform vec4 _SpecColor;
uniform vec4 unity_Ambient;
void main ()
{
  vec4 t0 = texture2D(_MainTex, gl_TexCoord[0].xy);
  vec3 t1 = (t0.xyz * _Color.xyz);
  float t2 = (t0.w * _Color.w);
  if ((t2 < _Cutoff)) {
    discard;
  }
  vec4 t3 = log2(texture2DProj(_LightBuffer, vec3(gl_TexCoord[1].xy, gl_TexCoord[1].w)));
  vec3 t4 = (unity_Ambient.xyz - t3.xyz);
  float t5 = (t3.w * t0.w);
  vec3 t6 = ((t1 * t4) - (t5 * (t4 * _SpecColor.xyz)));
  gl_FragData[0] = vec4(t6, (t2 - (t5 * _SpecColor.w)));
}
