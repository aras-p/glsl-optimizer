precision highp float;
precision highp int;
uniform mediump samplerExternalOES tex;
out mediump vec4 color;
in mediump vec4 uv;
void main ()
{
  mediump vec2 t0 = uv.xy;
  mediump float t1 = uv.z;
  mediump float t2 = uv.w;
  mediump vec4 t3 = (((((texture(tex, t0) + textureProj(tex, vec3(t0, t1))) + textureProj(tex, vec3(t0, t2))) + texture(tex, t0, t2)) + textureProj(tex, vec3(t0, t1), t2)) + textureProj(tex, vec3(t0, t2), t1));
  highp vec2 t4 = vec2(textureSize(tex, 0));

}
