// Changes done: removed Vulkan descriptor sets; made SSBO std430 layouts explicit;
// replaced push constants with a std140 UBO at binding 0; made atomic increment unsigned;
// selected GLSL 320 es; added default and image precision declarations.
precision highp float;
precision highp int;

layout (early_fragment_tests) in;

struct Node
{
    vec4 color;
    float depth;
    uint next;
};

layout (std430, binding = 1) buffer GeometrySBO
{
    uint count;
    uint maxNodeCount;
};

layout (binding = 2, r32ui) uniform coherent highp uimage2D headIndexImage;

layout (std430, binding = 3) buffer LinkedListSBO
{
    Node nodes[];
};

layout(std140, binding = 0) uniform PushConsts {
	mat4 model;
    vec4 color;
} pushConsts;

void main()
{
    // Increase the node count
    uint nodeIdx = atomicAdd(count, 1u);

    // Check LinkedListSBO is full
    if (nodeIdx < maxNodeCount)
    {
        // Exchange new head index and previous head index
        uint prevHeadIdx = imageAtomicExchange(headIndexImage, ivec2(gl_FragCoord.xy), nodeIdx);

        // Store node data
        nodes[nodeIdx].color = pushConsts.color;
        nodes[nodeIdx].depth = gl_FragCoord.z;
        nodes[nodeIdx].next = prevHeadIdx;
    }
}
