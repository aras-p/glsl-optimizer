#version 300 es
precision highp float;
precision highp int;
layout(std140) uniform Lighting {
  highp vec3 direction;
  highp float scale;
};
layout(std140) uniform Material {
  highp float unusedBefore;
  mediump vec4 tint;
  highp float unusedAfter;
};
uniform highp vec4 ordinary;
out highp vec4 color;
void main ()
{
  color = ((tint * vec4(direction, scale)) + ordinary);
}
