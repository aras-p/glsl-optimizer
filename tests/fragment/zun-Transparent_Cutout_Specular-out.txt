uniform vec4 _Color;
uniform float _Cutoff;
uniform vec4 _LightColor0;
uniform sampler2D _MainTex;
uniform float _Shininess;
uniform vec4 _SpecColor;
void main ()
{
  vec4 t0 = texture2D(_MainTex, gl_TexCoord[0].xy);
  vec3 t1 = (t0.xyz * _Color.xyz);
  float t2 = (t0.w * _Color.w);
  if ((t2 < _Cutoff)) {
    discard;
  }
  vec3 t3 = (gl_TexCoord[2].xyz + normalize(gl_TexCoord[3].xyz));
  vec3 t4 = (((t1 * _LightColor0.xyz) * max(0.0, dot(gl_TexCoord[1].xyz, gl_TexCoord[2].xyz))) + ((_LightColor0.xyz * _SpecColor.xyz) * (pow(max(0.0, dot(gl_TexCoord[1].xyz, normalize(t3))), (_Shininess * 128.0)) * t0.w)));
  vec3 t5 = (t4 * 2.0);
  gl_FragData[0] = vec4(t5, t2);
}
