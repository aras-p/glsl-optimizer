  highp mat3x2 columns[4];

  bvec2 enabled;
} lookup;
uniform highp int index;
out highp vec4 color;
void main ()
{

  highp uint t2 = (uint(index) & 1u);
  highp uint t3 = (uint(index) % 3u);

}
