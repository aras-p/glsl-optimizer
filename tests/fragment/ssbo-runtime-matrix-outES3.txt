#version 310 es
precision highp float;
precision highp int;
layout(std140, row_major) buffer Input {
  readonly highp float bias;
  readonly highp mat2x3 matrices[];
} source;
uniform highp int index;
out highp vec4 color;
void main ()
{
  highp vec3 t0 = ((source.matrices[index][0] + source.bias) + (source.matrices[index][1] * 2.0));
  color = vec4(t0, float(source.matrices.length()));
}
