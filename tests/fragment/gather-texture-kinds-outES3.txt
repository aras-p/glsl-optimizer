#version 320 es
precision highp float;
precision highp int;
uniform highp sampler2DArray layers;
uniform highp samplerCube cubeMap;
uniform highp samplerCubeArray cubeLayers;
uniform highp sampler2DArrayShadow shadowLayers;
uniform highp usampler2D labels;
in highp vec4 coord;
out highp vec4 color;
void main ()
{
  highp vec3 t0 = coord.xyz;
  highp vec4 t1 = (((textureGatherOffset(layers, t0, ivec2(1, -1), 2) + textureGather(cubeMap, t0)) + textureGather(cubeLayers, coord, 3)) + textureGather(shadowLayers, t0, coord.w));
  color = (t1 + vec4(textureGather(labels, coord.xy, 1)));
}

//  #3: shadowLayers (high 2dshadow) 0x0 [-1]
