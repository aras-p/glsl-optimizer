#version 320 es
uniform highp sampler2DArray layers;
uniform highp samplerCube cubeMap;
uniform highp samplerCubeArray cubeLayers;
uniform highp sampler2DArrayShadow shadowLayers;
uniform highp usampler2D labels;
in highp vec4 coord;
out highp vec4 color;
void main ()
{
  color = (((
    (textureGatherOffset (layers, coord.xyz, ivec2(1, -1), 2) + textureGather (cubeMap, coord.xyz, 0))
   + textureGather (cubeLayers, coord, 3)) + textureGather (shadowLayers, coord.xyz, coord.w)) + vec4(textureGather (labels, coord.xy, 1)));
}
