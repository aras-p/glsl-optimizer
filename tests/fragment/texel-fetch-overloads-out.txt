#version 330
uniform samplerBuffer buffer_texture;
uniform sampler2DRect rectangle_texture;
uniform sampler2DMS multisample_texture;
uniform int sample_index;
out vec4 color;
void main ()
{
  ivec2 t0 = ivec2(gl_FragCoord.xy);
  color = ((texelFetch(buffer_texture, t0.x) + texelFetch(rectangle_texture, t0)) + texelFetch(multisample_texture, t0, sample_index));
}

// uniforms: 1 (total size: 0)
//  #0: sample_index (high int) 1x1 [-1]
// textures: 3
//  #0: buffer_texture (high other) 0x0 [-1]
//  #1: rectangle_texture (high other) 0x0 [-1]
//  #2: multisample_texture (high other) 0x0 [-1]
