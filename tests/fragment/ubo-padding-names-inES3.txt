#version 300 es
precision highp float;
struct Entry {
    float _buf_pad1;
    vec4 color;
    float _buf_pad2;
    vec4 other;
};
layout(std140) uniform Data {
    float _buf_pad1;
    vec4 color;
    float _buf_pad2;
    Entry entry;
} data;
out vec4 color;
void main()
{
    color = data.color * data._buf_pad1 + data.entry.color * data.entry._buf_pad1
        + data.entry.other * (data._buf_pad2 + data.entry._buf_pad2);
}
