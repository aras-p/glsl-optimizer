uniform sampler2D _BumpMap;
uniform vec4 _Color;
uniform float _Cutoff;
uniform sampler2D _MainTex;
uniform float _Shininess;
void main ()
{
  vec2 t0 = ((texture2D(_BumpMap, gl_TexCoord[0].zw).wy * 2.0) - 1.0);
  float t1 = ((1.0 - (t0.x * t0.x)) - (t0.y * t0.y));
  if (((texture2D(_MainTex, gl_TexCoord[0].xy).w * _Color.w) < _Cutoff)) {
    discard;
  }
  vec3 t2 = vec3(t0, sqrt(t1));
  vec3 t3 = ((vec3(dot(gl_TexCoord[1].xyz, t2), dot(gl_TexCoord[2].xyz, t2), dot(gl_TexCoord[3].xyz, t2)) * vec3(0.5, 0.5, -0.5)) + 0.5);
  gl_FragData[0] = vec4(t3, _Shininess);
}
