#version 310 es
precision highp float;
precision highp int;
struct Entry {
  highp vec3 color;
  highp float weight;
  highp vec2 pairs[2];
};
layout(std140) buffer Data {
  highp int index;
  Entry entries[];
};
out highp vec4 color;
void main ()
{
  entries[index].weight = entries[index].pairs[1].x;
  highp vec3 t0 = (entries[index].color * entries[index].weight);
  color = vec4(t0, float(entries.length()));
}
