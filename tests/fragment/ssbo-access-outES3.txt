#version 310 es
precision highp float;
precision highp int;
layout(std430, binding = 3) buffer Output {
  mediump vec3 value;
  bvec3 flags;
  highp ivec2 indices;
  highp vec2 values[2][3];
} dest;
layout(std430, binding = 1) buffer Input {
  readonly mediump vec3 value;
  readonly bool enabled;
} source[2];
out highp vec4 color;
void main ()
{
  dest.value.xy = source[0].value.yxy.xy;
  dest.value.z = source[1].value.zzz.z;
  dest.flags = bvec3(source[1].enabled, true, false);
  highp ivec2 t0 = dest.indices;
  dest.values[t0.x][t0.y] = dest.value.xy;
  mediump vec3 t1 = dest.value;
  color = vec4(t1, float(dest.flags.x));
}
