uniform vec4 _Color;
uniform sampler2D _LightBuffer;
uniform sampler2D _MainTex;
uniform float _Parallax;
uniform sampler2D _ParallaxMap;
uniform vec4 unity_Ambient;
void main ()
{
  vec3 t0 = normalize(gl_TexCoord[1].xyz);
  vec4 t1 = (texture2D(_MainTex, (gl_TexCoord[0].xy + (((texture2D(_ParallaxMap, gl_TexCoord[0].zw).w * _Parallax) - (_Parallax / 2.0)) * (t0.xy / (t0.z + 0.42))))) * _Color);
  vec3 t2 = (t1.xyz * (unity_Ambient.xyz - log2(texture2DProj(_LightBuffer, vec3(gl_TexCoord[2].xy, gl_TexCoord[2].w))).xyz));
  gl_FragData[0] = vec4(t2, t1.w);
}
