#version 300 es
precision highp float;
precision highp int;
uniform highp sampler2D fullTexture;
uniform mediump sampler2D halfTexture;
in highp vec2 uv;
out highp vec4 result;
void main ()
{
  highp vec4 t0 = textureLod(fullTexture, uv, 0.0);
  highp vec4 t1 = vec4(lessThan(vec4(0.5), t0));
  highp float t2 = float(gl_FrontFacing);
  result = (((t0 + textureLod(halfTexture, uv, 0.0)) + t1) + t2);
}

//  #1: halfTexture (medium 2d) 0x0 [-1]
