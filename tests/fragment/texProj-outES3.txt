in highp vec4 uv;

{

  c_1 = (textureProj (tex, vec3(uv.xy, uv.w)) + textureProj (tex, vec3(uv.xy, uv.z)));

  c_1 = (c_1 + textureProjLod (tex, vec3(uv.xy, uv.z), 1.0));

  c_1 = (c_1 + vec4(textureProj (shadowmap, vec4(uv.xy, uv.z, uv.w))));

}
