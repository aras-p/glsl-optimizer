uniform vec4 _Color;
uniform samplerCube _Cube;
uniform sampler2D _LightBuffer;
uniform sampler2D _MainTex;
uniform vec4 _ReflectColor;
uniform vec4 unity_Ambient;
void main ()
{
  vec4 t0 = texture2D(_MainTex, gl_TexCoord[0].xy);
  vec4 t1 = (textureCube(_Cube, gl_TexCoord[1].xyz) * t0.w);
  vec3 t2 = (((t0 * _Color).xyz * (unity_Ambient.xyz - log2(texture2DProj(_LightBuffer, vec3(gl_TexCoord[2].xy, gl_TexCoord[2].w))).xyz)) + (t1.xyz * _ReflectColor.xyz));
  gl_FragData[0] = vec4(t2, (t1.w * _ReflectColor.w));
}
