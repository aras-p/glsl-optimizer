#version 300 es
precision highp float;
precision highp int;
layout(location = 0) out mediump vec4 _fragData;
uniform mediump sampler2D _MainTex;
uniform highp vec4 _TerrainTreeLightColors[4];
in highp vec2 xlv_uv;
in highp vec3 xlv_nl;
void main ()
{
  mediump vec4 t0 = texture(_MainTex, xlv_uv);
  if ((t0.w < 0.5)) {
    discard;
  }
  mediump vec3 t1 = ((t0.xyz * xlv_nl.x) * _TerrainTreeLightColors[0].xyz);
  mediump vec3 t2 = (t1 + ((t0.xyz * xlv_nl.y) * _TerrainTreeLightColors[1].xyz));
  mediump vec3 t3 = (t2 + ((t0.xyz * xlv_nl.z) * _TerrainTreeLightColors[2].xyz));
  _fragData = vec4(t3, 1.0);
}

// stats: 9 alu 2 tex 1 flow
// inputs: 2
//  #0: xlv_uv (high float) 2x1 [-1]
//  #1: xlv_nl (high float) 3x1 [-1]
// uniforms: 1 (total size: 0)
//  #0: _TerrainTreeLightColors (high float) 4x1 [4]
// textures: 1
//  #0: _MainTex (medium 2d) 0x0 [-1]
