uniform vec4 _Color;
uniform sampler2D _LightBuffer;
uniform sampler2D _MainTex;
uniform vec4 _SpecColor;
uniform sampler2D unity_Lightmap;
uniform sampler2D unity_LightmapInd;
void main ()
{
  vec4 t0 = texture2D(_MainTex, gl_TexCoord[0].xy);
  vec2 t1 = gl_TexCoord[2].xy;
  vec4 t2 = log2(texture2DProj(_LightBuffer, vec3(gl_TexCoord[1].xy, gl_TexCoord[1].w)));
  vec3 t3 = ((mix(texture2D(unity_LightmapInd, t1).xyz, texture2D(unity_Lightmap, t1).xyz, vec3(clamp(gl_TexCoord[2].z, 0.0, 1.0))) * 2.0) - t2.xyz);
  float t4 = (t2.w * t0.w);
  vec3 t5 = (((t0.xyz * _Color.xyz) * t3) - (t4 * (t3 * _SpecColor.xyz)));
  gl_FragData[0] = vec4(t5, ((t0.w * _Color.w) - (t4 * _SpecColor.w)));
}
