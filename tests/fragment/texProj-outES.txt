#extension GL_EXT_shader_texture_lod : enable
#extension GL_EXT_shadow_samplers : enable
mediump vec4 impl_medium_texture2DProjLodEXT(mediump sampler2D sampler, highp vec4 coord, mediump float lod)
{
#if defined(GL_EXT_shader_texture_lod)
	return texture2DProjLodEXT(sampler, coord, lod);
#else
	return texture2DProj(sampler, coord, lod);
#endif
}

mediump vec4 impl_medium_texture2DProjLodEXT(mediump sampler2D sampler, highp vec3 coord, mediump float lod)
{
#if defined(GL_EXT_shader_texture_lod)
	return texture2DProjLodEXT(sampler, coord, lod);
#else
	return texture2DProj(sampler, coord, lod);
#endif
}

uniform mediump sampler2D tex;
uniform mediump sampler2DShadow shadowmap;
varying highp vec4 uv;
void main ()
{
  mediump vec4 c_1;
  c_1 = (texture2DProj (tex, uv) + texture2DProj (tex, uv.xyz));
  c_1 = (c_1 + impl_medium_texture2DProjLodEXT (tex, uv, 1.0));
  c_1 = (c_1 + impl_medium_texture2DProjLodEXT (tex, uv.xyz, 1.0));
  c_1 = (c_1 + vec4(shadow2DEXT (shadowmap, uv.xyz)));
  c_1 = (c_1 + vec4(shadow2DProjEXT (shadowmap, uv)));
  gl_FragColor = c_1;
}
