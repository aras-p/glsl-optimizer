#extension GL_EXT_texture_array : enable
uniform sampler2DArray myarr;
varying vec4 uv;
void main ()
{
  vec3 t0 = uv.xyz;
  gl_FragData[0] = ((texture2DArray(myarr, t0) + texture2DArray(myarr, uv.xyw)) + texture2DArray(myarr, t0, 1.5));
}

// textures: 1
//  #0: myarr (high 2darray) 0x0 [-1]
