uniform vec4 _Color;
uniform sampler2D _Illum;
uniform sampler2D _LightBuffer;
uniform sampler2D _MainTex;
uniform vec4 _SpecColor;
uniform sampler2D unity_Lightmap;
uniform sampler2D unity_LightmapInd;
void main ()
{
  vec4 t0 = texture2D(_MainTex, gl_TexCoord[0].xy);
  vec4 t1 = (t0 * _Color);
  vec2 t2 = gl_TexCoord[2].xy;
  vec4 t3 = log2(texture2DProj(_LightBuffer, vec3(gl_TexCoord[1].xy, gl_TexCoord[1].w)));
  vec3 t4 = ((mix(texture2D(unity_LightmapInd, t2).xyz, texture2D(unity_Lightmap, t2).xyz, vec3(clamp(gl_TexCoord[2].z, 0.0, 1.0))) * 2.0) - t3.xyz);
  float t5 = (t3.w * t0.w);
  vec3 t6 = (((t1.xyz * t4) - (t5 * (t4 * _SpecColor.xyz))) + (t1.xyz * texture2D(_Illum, gl_TexCoord[0].zw).w));
  gl_FragData[0] = vec4(t6, (t1.w - (t5 * _SpecColor.w)));
}
