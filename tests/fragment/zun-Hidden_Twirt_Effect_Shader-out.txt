uniform sampler2D _MainTex;
uniform mat4 _RotationMatrix;
void main ()
{
  vec2 t0 = (gl_TexCoord[0].xy / _CenterRadius.zw);
  gl_FragData[0] = texture2D(_MainTex, (mix((_RotationMatrix * vec4(gl_TexCoord[0].xy, 0.0, 0.0)).xy, gl_TexCoord[0].xy, vec2(min(1.0, length(t0)))) + _CenterRadius.xy));
}
