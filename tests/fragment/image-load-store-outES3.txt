#version 310 es
precision highp float;
precision highp int;
layout(rgba8, binding = 1) readonly uniform highp image3D sourceImage;
layout(rgba32ui, binding = 3) writeonly uniform highp uimage2DArray destinationImage;
layout(location = 0) out highp vec4 color;
void main ()
{
  mediump ivec2 t0 = ivec2(gl_FragCoord.xy);
  highp vec4 t1 = imageLoad(sourceImage, ivec4(t0, (imageSize(sourceImage).z - 1), 0).xyz);
  color = t1;
  imageStore(destinationImage, ivec4(t0, (imageSize(destinationImage).z - 1), 0).xyz, uvec4((t1 * 255.0)));
}

// inputs: 1
//  #0: gl_FragCoord (high float) 4x1 [-1] loc 0
// textures: 2
//  #0: sourceImage (high image3d) 0x0 [-1] loc 1 format rgba8 component float readonly
//  #1: destinationImage (high image2darray) 0x0 [-1] loc 3 format rgba32ui component uint writeonly
