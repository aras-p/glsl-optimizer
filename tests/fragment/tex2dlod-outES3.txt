#version 300 es
precision highp float;
precision highp int;
uniform mediump sampler2D tex;
in highp vec4 uvHi;
in mediump vec4 uvMed;
out mediump vec4 _fragColor;
void main ()
{
  mediump vec4 t0 = vec4(uvHi.xy, 0.0, 0.0);
  _fragColor = (textureLod(tex, t0.xy, t0.w) + textureLod(tex, uvMed.xy, uvMed.z));
}

//  #1: uvMed (medium float) 4x1 [-1]
// textures: 1
