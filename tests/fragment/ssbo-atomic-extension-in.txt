#version 420
#extension GL_ARB_shader_storage_buffer_object : require
layout(std140, binding=0) buffer Data { ivec4 values[]; };
uniform int index;
out vec4 color;
void main()
{
    int old = atomicAdd(values[index].x, 1);
    atomicCompSwap(values[index].y, old, 2);
    color = vec4(float(old));
}
