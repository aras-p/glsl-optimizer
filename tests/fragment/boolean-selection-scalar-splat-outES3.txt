#version 300 es
precision highp float;
precision highp int;
uniform highp vec3 a;
uniform highp vec3 b;
uniform highp float value;
uniform bvec3 mask;
out highp vec4 color;
void main ()
{
  highp vec3 t0 = (mix(a, b, mask) + ((value == 0.0) ? b : a));
  color = vec4(t0, 1.0);
}

//  #1: b (high float) 3x1 [-1]
//  #2: value (high float) 1x1 [-1]
//  #3: mask (high bool) 3x1 [-1]
