uniform sampler2D _LightBuffer;
uniform sampler2D _MainTex;
uniform vec4 unity_Ambient;
void main ()
{
  if ((fract(((gl_TexCoord[1].y + (gl_TexCoord[1].z * 0.1)) * 5.0)) < 0.5)) {
    discard;
  }
  vec3 t0 = (texture2D(_MainTex, gl_TexCoord[0].xy).xyz * (unity_Ambient.xyz - log2(texture2DProj(_LightBuffer, vec3(gl_TexCoord[2].xy, gl_TexCoord[2].w))).xyz));
  gl_FragData[0] = vec4(t0, 0.0);
}
