#version 320 es
precision highp float;
precision highp int;
struct Node {
  highp vec4 color;
  highp float depth;
  highp uint next;
};
layout(std430, binding = 1) buffer LinkedListSBO {
  Node nodes[];
};
layout(location = 0) out highp vec4 outFragColor;
layout(r32ui, binding = 0) uniform highp uimage2D headIndexImage;
void main ()
{
  highp uint r0;
  highp int r1;
  highp int r2;
  highp int r3;
  highp int r4;
  highp int r5;
  highp vec4 r6;
  highp vec4 fragments_color[128];
  highp float fragments_depth[128];
  highp uint fragments_next[128];
  highp ivec2 t0 = ivec2(gl_FragCoord.xy);
  r0 = imageLoad(headIndexImage, ivec4(t0, 0, 0).xy).x;
  for (r4 = 0; ((int(r0) != -1) && (r4 < 128)); r4++) {
    fragments_color[r4] = nodes[int(r0)].color;
    fragments_depth[r4] = nodes[int(r0)].depth;
    r0 = nodes[int(r0)].next;
    fragments_next[r4] = r0;
  }
  for (r3 = 1; (uint(r3) < uint(r4)); r3++) {
    highp vec4 t1 = fragments_color[r3];
    highp float t2 = fragments_depth[r3];
    highp uint t3 = fragments_next[r3];
    r2 = r3;
    while (true) {
      r1 = (r2 - 1);
      highp float t4 = fragments_depth[r1];
      if ((!((r2 != 0) && (t4 < t2)))) {
        break;
      }
      fragments_color[r2] = fragments_color[r1];
      fragments_depth[r2] = t4;
      fragments_next[r2] = fragments_next[r1];
      r2 = r1;
    }
    fragments_color[r2] = t1;
    fragments_depth[r2] = t2;
    fragments_next[r2] = t3;
  }
  r5 = 0;
  r6 = vec4(0.025, 0.025, 0.025, 1.0);
  for (; (r5 < r4); r5++) {
    highp vec4 t5 = fragments_color[r5];
    r6 = mix(r6, t5, t5.wwww);
  }
  outFragColor = r6;
}
