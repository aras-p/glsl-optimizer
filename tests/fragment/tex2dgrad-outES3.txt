uniform mediump samplerCube cub;
in mediump vec3 uv1;
in highp vec3 uv2;
out mediump vec4 _fragColor;
void main ()
{
  _fragColor = (((textureGrad (tex, uv1.xy, 
    dFdx(uv1.xy)
  , 
    dFdy(uv1.xy)
  ) + textureGrad (tex, uv2.xy, 
    dFdx(uv2.xy)
  , 
    dFdy(uv2.xy)
  )) + textureGrad (tex, uv2.xy, uv1.xy, uv1.xy)) + textureGrad (cub, uv2, dFdx(uv2), dFdy(uv2)));
}
