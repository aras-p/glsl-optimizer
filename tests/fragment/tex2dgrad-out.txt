#extension GL_ARB_shader_texture_lod : enable
uniform sampler2D tex;
uniform samplerCube cub;
varying vec3 uv;
void main ()
{
  vec2 t0 = uv.xy;
  gl_FragColor = (texture2DGradARB(tex, t0, dFdx(t0), dFdy(t0)) + textureCubeGradARB(cub, uv, dFdx(uv), dFdy(uv)));
}

// stats: 1 alu 2 tex 0 flow
// inputs: 1
//  #0: uv (high float) 3x1 [-1]
// textures: 2
//  #0: tex (high 2d) 0x0 [-1]
//  #1: cub (high cube) 0x0 [-1]
