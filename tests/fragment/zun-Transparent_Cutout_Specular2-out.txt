uniform vec4 _Color;
uniform float _Cutoff;
uniform sampler2D _LightBuffer;
uniform sampler2D _MainTex;
uniform vec4 _SpecColor;
uniform sampler2D unity_Lightmap;
uniform sampler2D unity_LightmapInd;
void main ()
{
  vec4 t0 = texture2D(_MainTex, gl_TexCoord[0].xy);
  vec3 t1 = (t0.xyz * _Color.xyz);
  float t2 = (t0.w * _Color.w);
  if ((t2 < _Cutoff)) {
    discard;
  }
  vec2 t3 = gl_TexCoord[2].xy;
  vec4 t4 = log2(texture2DProj(_LightBuffer, vec3(gl_TexCoord[1].xy, gl_TexCoord[1].w)));
  vec3 t5 = ((mix(texture2D(unity_LightmapInd, t3).xyz, texture2D(unity_Lightmap, t3).xyz, vec3(clamp(gl_TexCoord[2].z, 0.0, 1.0))) * 2.0) - t4.xyz);
  float t6 = (t4.w * t0.w);
  vec3 t7 = ((t1 * t5) - (t6 * (t5 * _SpecColor.xyz)));
  gl_FragData[0] = vec4(t7, (t2 - (t6 * _SpecColor.w)));
}
