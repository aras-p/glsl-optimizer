#version 300 es
struct Values {
  mediump vec4 color;
  highp vec4 position;
};
uniform Values values;
uniform mediump sampler2D tex;
uniform mediump vec4 tint;
uniform mediump vec4 scale;
uniform highp vec4 offset;
in highp vec2 uv;
out mediump vec4 fragColor;
void main ()
{
  fragColor = (((
    ((values.color * tint) * scale)
   + 0.25) * texture (tex, uv)) + (values.position + offset));
}
