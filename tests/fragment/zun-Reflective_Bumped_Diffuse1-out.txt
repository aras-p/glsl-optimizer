uniform sampler2D _BumpMap;
uniform vec4 _Color;
uniform samplerCube _Cube;
uniform sampler2D _LightBuffer;
uniform sampler2D _MainTex;
uniform vec4 _ReflectColor;
uniform sampler2D unity_Lightmap;
uniform sampler2D unity_LightmapInd;
void main ()
{
  vec4 t0 = texture2D(_MainTex, gl_TexCoord[0].xy);
  vec2 t1 = ((texture2D(_BumpMap, gl_TexCoord[0].zw).wy * 2.0) - 1.0);
  vec3 t2 = vec3(t1, sqrt(((1.0 - (t1.x * t1.x)) - (t1.y * t1.y))));
  vec3 t3 = vec3(dot(gl_TexCoord[2].xyz, t2), dot(gl_TexCoord[3].xyz, t2), dot(gl_TexCoord[4].xyz, t2));
  vec3 t4 = vec3(gl_TexCoord[2].w, gl_TexCoord[3].w, gl_TexCoord[4].w);
  vec4 t5 = (textureCube(_Cube, (t4 - ((vec3(dot(t3, t4)) * 2.0) * t3))) * t0.w);
  vec2 t6 = gl_TexCoord[5].xy;
  vec3 t7 = ((mix(texture2D(unity_LightmapInd, t6).xyz, texture2D(unity_Lightmap, t6).xyz, vec3(clamp(gl_TexCoord[5].z, 0.0, 1.0))) * 2.0) - log2(texture2DProj(_LightBuffer, vec3(gl_TexCoord[1].xy, gl_TexCoord[1].w))).xyz);
  vec3 t8 = (((t0 * _Color).xyz * t7) + (t5.xyz * _ReflectColor.xyz));
  gl_FragData[0] = vec4(t8, (t5.w * _ReflectColor.w));
}
