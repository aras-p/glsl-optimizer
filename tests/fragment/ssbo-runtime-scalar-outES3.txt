#version 310 es
precision highp float;
precision highp int;
layout(std430) buffer Data {
  highp vec3 header;
  highp float values[];
} data;
uniform highp int index;
out highp vec4 color;
void main ()
{
  highp int t0 = data.values.length();
  if ((index < t0)) {
    highp float t1 = (data.values[index] + data.header.x);
    data.values[index] = t1;
    data.header.y = vec3(t1).y;
  }
  highp vec3 t2 = data.header;
  color = vec4(t2, float(t0));
}
