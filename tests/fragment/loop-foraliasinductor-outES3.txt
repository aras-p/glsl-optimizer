#version 300 es
precision highp float;
precision highp int;
layout(location = 0) out mediump vec4 _fragData;
uniform mediump sampler2D _MainTex;
in highp vec2 xlv_uv;
void main ()
{
  highp float r0;
  mediump vec3 r1;
  mediump vec4 t0 = texture(_MainTex, xlv_uv);
  if ((t0.x >= 0.0)) {
    r1 = vec3(0.0);
    for (r0 = 0.0; (r0 < xlv_uv.x); r0 += 1.0) {
      r1 = (r1 + t0.xyz);
    }
  } else {
    r1 = vec3(0.0);
  }
  mediump float t1 = xlv_uv.y;
  mediump vec3 t2 = ((t0.x >= 1.0) ? (r1 + (t0.xyz * (t1 * sin(t1)))) : r1);
  _fragData = vec4(t2, 1.0);
}
