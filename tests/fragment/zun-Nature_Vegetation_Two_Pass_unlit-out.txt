uniform vec4 _Color;
uniform float _Cutoff;
uniform sampler2D _MainTex;
void main ()
{
  vec4 t0 = (_Color * texture2D(_MainTex, gl_TexCoord[0].xy));
  if ((t0.w < _Cutoff)) {
    discard;
  }
  gl_FragData[0] = t0;
}
