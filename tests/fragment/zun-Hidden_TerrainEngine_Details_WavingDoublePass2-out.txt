uniform float _Cutoff;
uniform sampler2D _LightBuffer;
uniform sampler2D _MainTex;
uniform vec4 unity_Ambient;
void main ()
{
  vec4 t0 = (texture2D(_MainTex, gl_TexCoord[0].xy) * gl_Color);
  if ((t0.w < _Cutoff)) {
    discard;
  }
  vec3 t1 = (t0.xyz * (unity_Ambient.xyz - log2(texture2DProj(_LightBuffer, vec3(gl_TexCoord[1].xy, gl_TexCoord[1].w))).xyz));
  gl_FragData[0] = vec4(t1, t0.w);
}
