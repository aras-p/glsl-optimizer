#version 300 es
precision highp float;
precision highp int;
in mediump vec4 a;
in mediump vec4 b;
layout(location = 0) out mediump vec4 andMask;
layout(location = 1) out mediump vec4 orMask;
layout(location = 2) out mediump vec4 xorMask;
layout(location = 3) out mediump vec4 notAndMask;
void main ()
{
  bvec4 t0 = lessThan(a, b);
  bvec4 t1 = greaterThanEqual(a, vec4(0.0));
  andMask = vec4(bvec4((t0.x && t1.x), (t0.y && t1.y), (t0.z && t1.z), (t0.w && t1.w)));
  notAndMask = vec4(not(bvec4((t0.x && t1.x), (t0.y && t1.y), (t0.z && t1.z), (t0.w && t1.w))));
  orMask = vec4(bvec4((t0.x || t1.x), (t0.y || t1.y), (t0.z || t1.z), (t0.w || t1.w)));
  xorMask = vec4(bvec4((t0.x ^^ t1.x), (t0.y ^^ t1.y), (t0.z ^^ t1.z), (t0.w ^^ t1.w)));
}
