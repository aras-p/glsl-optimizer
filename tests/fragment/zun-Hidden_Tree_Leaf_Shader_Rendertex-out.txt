uniform float _HalfOverCutoff;
uniform sampler2D _MainTex;
void main ()
{
  vec4 t0 = texture2D(_MainTex, gl_TexCoord[0].xy);
  vec3 t1 = (t0.xyz * (gl_TexCoord[1].xyz * 2.0));
  float t2 = (t0.w * (2.0 * _HalfOverCutoff));
  if ((t2 < 1.0)) {
    discard;
  }
  gl_FragData[0] = vec4(t1, t2);
}
