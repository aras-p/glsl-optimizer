#version 450
in Payload {
  vec2 value;
} payloadIn;
layout(location = 14) in Anonymous {
  vec4 unused;
  flat int id;
};
layout(location = 4) in Extra {
  noperspective vec2 uv;
  mat2 basis;
} extraIn[2];
layout(location = 0) out vec4 color;
void main ()
{
  vec2 t0 = extraIn[int((uint(id) & 1u))].uv;
  vec2 t1 = (extraIn[int((uint(id) & 1u))].basis * t0);
  color = vec4(t1, payloadIn.value);
}

// inputs: 3
//  #0: payloadIn (high other) 0x0 [-1]
//  #1: id (high int) 1x1 [-1] loc 14
//  #2: extraIn (high other) 0x0 [2] loc 4
