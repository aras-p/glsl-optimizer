#version 310 es
precision highp float;
precision highp int;
layout(std430) buffer Data {
  highp uint value;
  highp uint count;
} data;
uniform bool update;
out highp vec4 color;
void main ()
{
  highp uint t0 = data.value;
  data.value = 7u;
  memoryBarrier();
  highp uint t1 = data.value;
  if (update) {
    atomicAdd(data.count, 1u);
    memoryBarrier();
    data.value = data.count;
  }
  memoryBarrier();
  color = vec4(float(t0), float(t1), float(data.value), 1.0);
}
