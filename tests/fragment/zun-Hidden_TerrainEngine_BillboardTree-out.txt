uniform sampler2D _MainTex;
void main ()
{
  vec4 t0 = texture2D(_MainTex, gl_TexCoord[0].xy);
  vec3 t1 = (t0.xyz * gl_Color.xyz);
  if ((t0.w < 0.0)) {
    discard;
  }
  gl_FragData[0] = vec4(t1, t0.w);
}
