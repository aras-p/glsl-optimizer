uniform sampler2D maintex;
uniform float factor;
varying vec2 uv;
void main ()
{
  vec4 t0 = texture2D(maintex, uv);
  vec2 t1 = ((t0.w >= 0.5) ? vec2(t0.x, (t0.y + factor)) : vec2((t0.x + factor), t0.y));
  gl_FragColor = vec4(t1, 0.0, 0.0);
}

// uniforms: 1 (total size: 0)
//  #0: factor (high float) 1x1 [-1]
// textures: 1
//  #0: maintex (high 2d) 0x0 [-1]
