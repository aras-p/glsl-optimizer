uniform sampler2D _BumpMap;
uniform float _Shininess;
void main ()
{
  vec2 t0 = ((texture2D(_BumpMap, gl_TexCoord[0].xy).wy * 2.0) - 1.0);
  vec3 t1 = vec3(t0, sqrt(((1.0 - (t0.x * t0.x)) - (t0.y * t0.y))));
  vec3 t2 = ((vec3(dot(gl_TexCoord[1].xyz, t1), dot(gl_TexCoord[2].xyz, t1), dot(gl_TexCoord[3].xyz, t1)) * vec3(0.5, 0.5, -0.5)) + 0.5);
  gl_FragData[0] = vec4(t2, _Shininess);
}
