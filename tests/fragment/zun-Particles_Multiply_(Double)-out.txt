uniform sampler2D _MainTex;
void main ()
{
  vec4 t0 = texture2D(_MainTex, gl_TexCoord[0].xy);
  vec3 t1 = ((t0.xyz * 2.0) * gl_Color.xyz);
  float t2 = (gl_Color.w * t0.w);
  gl_FragData[0] = mix(vec4(0.5), vec4(t1, t2), vec4(t2));
}
