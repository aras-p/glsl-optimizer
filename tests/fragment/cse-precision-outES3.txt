#version 300 es
precision highp float;
precision highp int;
uniform mediump sampler2D mediumTex;
uniform highp sampler2D highTex;
in highp vec2 uv;
layout(location = 0) out highp vec4 result;
void main ()
{
  highp vec4 t0 = texture(highTex, uv);
  mediump vec4 t1 = t0;
  result = (texture(mediumTex, uv) * (t0 + t1));
}

//  #0: uv (high float) 2x1 [-1]
// textures: 2
//  #0: mediumTex (medium 2d) 0x0 [-1]
//  #1: highTex (high 2d) 0x0 [-1]
