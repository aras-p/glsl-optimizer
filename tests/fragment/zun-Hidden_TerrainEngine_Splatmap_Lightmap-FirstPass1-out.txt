{
  vec4 t0 = texture2D(_Control, gl_TexCoord[0].xy);
  vec3 t1 = ((mat3(texture2D(_Splat0, gl_TexCoord[0].zw).xyz, texture2D(_Splat1, gl_TexCoord[1].xy).xyz, texture2D(_Splat2, gl_TexCoord[1].zw).xyz) * t0.xyz) + (t0.w * texture2D(_Splat3, gl_TexCoord[2].xy).xyz));
  vec2 t2 = gl_TexCoord[4].xy;
  vec3 t3 = ((mix(texture2D(unity_LightmapInd, t2).xyz, texture2D(unity_Lightmap, t2).xyz, vec3(clamp(gl_TexCoord[4].z, 0.0, 1.0))) * 2.0) - log2(texture2DProj(_LightBuffer, vec3(gl_TexCoord[3].xy, gl_TexCoord[3].w))).xyz);

}
