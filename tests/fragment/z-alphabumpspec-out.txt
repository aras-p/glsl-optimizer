uniform sampler2D _BumpMap;
uniform vec4 _Color;
uniform vec4 _LightColor0;
uniform sampler2D _MainTex;
uniform float _Shininess;
uniform vec4 _SpecColor;
void main ()
{
  vec4 t0 = texture2D(_MainTex, gl_TexCoord[0].xy);
  vec3 t1 = (t0.xyz * _Color.xyz);
  vec2 t2 = ((texture2D(_BumpMap, gl_TexCoord[0].zw).wy * 2.0) - 1.0);
  vec3 t3 = (gl_TexCoord[2].xyz + normalize(gl_TexCoord[1].xyz));
  vec3 t4 = vec3(t2, sqrt(((1.0 - (t2.x * t2.x)) - (t2.y * t2.y))));
  vec3 t5 = (((t1 * _LightColor0.xyz) * max(0.0, dot(t4, gl_TexCoord[2].xyz))) + ((_LightColor0.xyz * _SpecColor.xyz) * (pow(max(0.0, dot(t4, normalize(t3))), (_Shininess * 128.0)) * t0.w)));
  vec3 t6 = ((t5 * 2.0) + (t1 * gl_TexCoord[3].xyz));
  gl_FragData[0] = vec4(t6, (t0.w * _Color.w));
}
