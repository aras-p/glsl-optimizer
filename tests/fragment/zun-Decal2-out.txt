uniform vec4 _Color;
uniform sampler2D _DecalTex;
uniform sampler2D _LightBuffer;
uniform sampler2D _MainTex;
uniform vec4 unity_Ambient;
void main ()
{
  vec4 t0 = texture2D(_MainTex, gl_TexCoord[0].xy);
  vec4 t1 = texture2D(_DecalTex, gl_TexCoord[0].zw);
  vec3 t2 = mix(t0.xyz, t1.xyz, t1.www);
  vec4 t3 = (vec4(t2, t0.w) * _Color);
  vec3 t4 = (t3.xyz * (unity_Ambient.xyz - log2(texture2DProj(_LightBuffer, vec3(gl_TexCoord[1].xy, gl_TexCoord[1].w))).xyz));
  gl_FragData[0] = vec4(t4, t3.w);
}
