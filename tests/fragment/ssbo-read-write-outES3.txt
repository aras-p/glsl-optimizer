#version 310 es
precision highp float;
precision highp int;
struct Entry {
  highp vec3 color;
  highp float weight;
};
layout(std140, binding = 1) uniform Params {
  highp vec4 tint;
} params;
layout(std430, binding = 3) buffer Data {
  highp uint index;
  highp float values[3];
  Entry entries[2];
  highp vec2 pairs[2];
  highp uint unused;
} data;
out highp vec4 color;
void main ()
{
  highp uint t0 = (data.index & 1u);
  highp float t1 = data.values[int(t0)];
  data.values[int(t0)] = (t1 + 1.0);
  data.entries[int(t0)].color = params.tint.xyz;
  data.entries[int(t0)].weight = t1;
  data.pairs[int(t0)] = vec2(t1, data.values[int(t0)]);
  highp vec3 t2 = (data.entries[int(t0)].color * data.entries[int(t0)].weight);
  color = vec4(t2, data.values[int(t0)]);
}
