#version 430
#extension GL_KHR_shader_subgroup_arithmetic : enable
#extension GL_KHR_shader_subgroup_ballot : enable
#extension GL_KHR_shader_subgroup_basic : enable
flat in ivec2 range;
flat in uint mask;
layout(location = 0) out uvec4 result;
void main ()
{
  int t0 = subgroupMin(range.x);
  int t1 = subgroupBroadcastFirst(t0);
  int t2 = subgroupMax(range.y);
  int t3 = subgroupBroadcastFirst(t2);
  uint t4 = subgroupOr(mask);
  uint t5 = subgroupBroadcastFirst(t4);
  ivec2 t6 = subgroupMax(range);
  result = uvec4(ivec4(t1, t3, int(t5), t6.y));
}
