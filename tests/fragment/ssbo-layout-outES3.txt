#version 310 es
precision highp float;
precision highp int;
struct Pair {
  highp float x;
  highp float y;
};
layout(std140, binding = 4) buffer Output {
  writeonly highp float values[2];
  writeonly highp vec4 result;
};
layout(std430, row_major, binding = 2) buffer Input {
  readonly Pair pairs[2];
  readonly highp mat2 matrices[2];
  layout(column_major) readonly highp mat2x3 transform;
  readonly highp vec3 direction;
  readonly highp float weight;
} source;
uniform highp int index;
out highp vec4 color;
void main ()
{
  highp float t0 = source.pairs[index].x;
  highp float t1 = source.pairs[index].y;
  highp vec2 t2 = (source.matrices[index] * vec2(t0, t1));
  values[index] = t2.x;
  highp vec3 t3 = (((source.transform[0] * t2.x) + source.direction) + (source.transform[1] * t2.y));
  result = vec4(t3, source.weight);
  color = vec4(t2, t0, t1);
}
