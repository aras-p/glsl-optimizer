#version 320 es
precision highp float;
precision highp int;
layout(std430, binding = 3) restrict buffer ClusterRender {
  restrict highp uint data[];
} cluster_render;
layout(std140, binding = 1) uniform State {
  highp mat4 projection;
  highp float inv_z_far;
  highp uint screen_to_clusters_shift;
  highp uint cluster_screen_width;
  highp uint cluster_data_size;
  highp uint cluster_depth_offset;
  highp uint pad0;
  highp uint pad1;
  highp uint pad2;
} state;
layout(location = 0) in highp float depth_interp;
layout(location = 1) flat in highp uint element_index;
layout(location = 0) out highp vec4 frag_color;
void main ()
{
  highp uvec2 t0 = (uvec2(gl_FragCoord.xy) >> state.screen_to_clusters_shift);
  highp int t1 = ((int(t0.x) + (int(state.cluster_screen_width) * int(t0.y))) * int(state.cluster_data_size));
  atomicOr(cluster_render.data[(t1 + int((element_index >> 5u)))], uint((1 << element_index)));
  highp uint t2 = atomicOr(cluster_render.data[((t1 + int(state.cluster_depth_offset)) + int(element_index))], uint((1 << uint(clamp(floor(((depth_interp * 32.0) * state.inv_z_far)), 0.0, 31.0)))));
  frag_color = vec4(float(t2));
}
