uniform vec4 _Tint;
uniform sampler2D _UpTex;
void main ()
{
  vec4 t0 = texture2D(_UpTex, gl_TexCoord[0].xy);
  vec3 t1 = ((t0.xyz - 0.5) + _Tint.xyz);
  gl_FragData[0] = vec4(t1, (t0.w * _Tint.w));
}
