#version 450
layout(std430, binding = 2) buffer Data {
  layout(offset = 16) vec2 bias;
  float values[];
} data;
uniform int index;
out vec4 color;
void main ()
{
  float t0 = (data.values[index] + data.bias.x);
  data.values[index] = t0;
  vec2 t1 = data.bias;
  color = vec4(t0, float(data.values.length()), t1);
}
