#version 320 es
precision highp float;
precision highp int;
layout(early_fragment_tests) in;
struct Node {
  highp vec4 color;
  highp float depth;
  highp uint next;
};
layout(std140, binding = 0) uniform PushConsts {
  highp mat4 model;
  highp vec4 color;
} pushConsts;
layout(std430, binding = 3) buffer LinkedListSBO {
  Node nodes[];
};
layout(std430, binding = 1) buffer GeometrySBO {
  highp uint count;
  highp uint maxNodeCount;
};
layout(r32ui, binding = 2) coherent uniform highp uimage2D headIndexImage;
void main ()
{
  highp uint t0 = atomicAdd(count, 1u);
  if ((t0 < maxNodeCount)) {
    highp ivec2 t1 = ivec2(gl_FragCoord.xy);
    highp uint t2 = imageAtomicExchange(headIndexImage, ivec4(t1, 0, 0).xy, t0);
    nodes[int(t0)].color = pushConsts.color;
    nodes[int(t0)].depth = gl_FragCoord.z;
    nodes[int(t0)].next = t2;
  }
}
