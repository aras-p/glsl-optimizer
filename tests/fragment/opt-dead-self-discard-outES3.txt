#version 300 es
precision highp float;
precision highp int;
uniform highp float value;
out highp vec4 color;
void main ()
{
  highp int r0;
  highp float r1;
  r1 = value;
  for (r0 = 0; (r0 < 32); r0++) {
    r1 = (r1 * 1.01);
  }
  highp float t0 = (r1 + 0.25);
  if ((t0 < 0.0)) {
    discard;
  }
  color = vec4(t0);
}
