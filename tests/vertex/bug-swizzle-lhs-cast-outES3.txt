#version 300 es
in highp vec4 _glesVertex;
uniform highp vec3 _WorldSpaceCameraPos;
uniform highp mat4 glstate_matrix_mvp;
uniform highp mat4 _World2Object;
uniform highp vec3 _TerrainTreeLightDirections[4];
out highp vec3 xlv_TEXCOORD2;
void main ()
{
  highp vec3 t0 = ((_World2Object * vec4(_WorldSpaceCameraPos, 1.0)).xyz - _glesVertex.xyz);
  highp vec3 t1 = normalize(t0);
  mediump float t2 = clamp(dot(t1, (-_TerrainTreeLightDirections[0])), 0.0, 1.0);
  mediump float t3 = clamp(dot(t1, (-_TerrainTreeLightDirections[1])), 0.0, 1.0);
  mediump float t4 = clamp(dot(t1, (-_TerrainTreeLightDirections[2])), 0.0, 1.0);
  gl_Position = (glstate_matrix_mvp * _glesVertex);
  xlv_TEXCOORD2 = (2.0 * mix(mix(vec3(t2), vec3(t3), bvec3(false, true, false)), vec3(t4), bvec3(false, false, true)));
}
