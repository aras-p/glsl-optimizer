#version 310 es
struct Light {
  highp vec3 direction;
  highp float intensity;
};
layout(std140, row_major, binding = 3) uniform Scene {
  highp float unused;
  mediump mat2x3 transform;
  layout(column_major) highp mat3x2 other;
  Light lights[2];
  mediump float weights[3];
  bool enabled;
  highp uint flags;
  mediump int count;
} scene;
uniform highp int index;
void main ()
{
  highp float r0;
  highp vec3 t0 = (scene.transform * vec2(scene.weights[index], 1.0));
  highp vec3 t1 = (t0 + (scene.lights[index].direction * scene.lights[1].intensity));
  highp vec2 t2 = (t1.xy + (scene.other * vec3(1.0, 1.0, 1.0)));
  highp vec3 t3 = (vec3(t2, t1.z) * float(scene.count));
  if (scene.enabled) {
    r0 = float(scene.flags);
  } else {
    r0 = 0.0;
  }
  gl_Position = vec4(t3, r0);
}
