#version 310 es
struct Inner {
  highp vec3 direction;
  highp float scale;
  highp mat2 basis;
};
struct Outer {
  Inner inner;
  highp vec2 pairs[2];
};
layout(std140, row_major) uniform Data {
  highp float table[2][3];
  Outer items[2];
} data;
uniform highp int index;
void main ()
{
  highp vec2 value_pairs[2];
  value_pairs[0] = data.items[index].pairs[0];
  value_pairs[1] = data.items[index].pairs[1];
  highp vec2 t0 = value_pairs[index];
  highp vec2 t1 = (data.items[index].inner.basis * t0);
  gl_Position = vec4(t1, data.table[index][2], data.items[index].inner.scale);
}
