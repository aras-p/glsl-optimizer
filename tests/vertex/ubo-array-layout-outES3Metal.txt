  , constant _xlat_buf_Data_std140_row& data [[buffer(1)]])
{
  xlatMtlShaderOutput _mtl_o;
  float2 value_pairs[2];
  value_pairs[0] = float2(data.items[_mtl_u.index].pairs[0].value);
  value_pairs[1] = float2(data.items[_mtl_u.index].pairs[1].value);
  float2 t0 = value_pairs[_mtl_u.index];

  return _mtl_o;
}
