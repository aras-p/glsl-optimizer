#version 320 es
struct RenderElement {
  highp uint type;
  bool touches_near;
  bool touches_far;
  highp uint original_index;
  highp mat3x4 transform_inv;
  highp vec3 scale;
  highp uint pad;
};
layout(std430, column_major, binding=2) buffer RenderElements {
  layout(column_major) readonly restrict RenderElement data[];
} render_elements;
layout(std140, column_major, binding=1) uniform State {
  layout(column_major) highp mat4 projection;
  highp float inv_z_far;
  highp uint screen_to_clusters_shift;
  highp uint cluster_screen_width;
  highp uint cluster_data_size;
  highp uint cluster_depth_offset;
  highp uint pad0;
  highp uint pad1;
  highp uint pad2;
} state;
layout(std140, column_major, binding=0) uniform Params {
  highp uint base_index;
  highp uint pad0;
  highp uint pad1;
  highp uint pad2;
} params;
layout(location=0) in highp vec3 vertex_attrib;
layout(location=0) out highp float depth_interp;
layout(location=1) flat out highp uint element_index;
void main ()
{
  highp vec3 vertex_1;
  element_index = (params.base_index + uint(gl_InstanceID));
  vertex_1 = (vertex_attrib * render_elements.data[element_index].scale);
  highp vec4 tmpvar_2;
  tmpvar_2.w = 1.0;
  tmpvar_2.xyz = vertex_1;
  vertex_1 = (tmpvar_2 * render_elements.data[element_index].transform_inv);
  depth_interp = -(vertex_1.z);
  highp vec4 tmpvar_3;
  tmpvar_3.w = 1.0;
  tmpvar_3.xyz = vertex_1;
  gl_Position = (state.projection * tmpvar_3);
}
