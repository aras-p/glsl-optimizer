#version 310 es
layout(local_size_x=8) in;
layout(r32ui, binding=0) uniform readonly highp uimage2D source2D;
layout(r32i, binding=1) uniform readonly highp iimage2DArray sourceArray;
layout(r32f, binding=2) uniform readonly highp image3D source3D;
layout(std430) buffer Output { vec4 values[]; } outputData;
void main()
{
    int x = int(gl_GlobalInvocationID.x) - 1;
    uvec4 a = imageLoad(source2D, ivec2(x, imageSize(source2D).y));
    ivec4 b = imageLoad(sourceArray, ivec3(x, 0, imageSize(sourceArray).z));
    vec4 c = imageLoad(source3D, ivec3(x, 0, imageSize(source3D).z));
    outputData.values[gl_GlobalInvocationID.x] = vec4(a) + vec4(b) + c;
}
