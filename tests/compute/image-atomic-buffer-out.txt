#version 430
layout(local_size_x = 8) in;
layout(std430) buffer Results {
  uint values[];
} results;
layout(r32ui, binding = 3) uniform uimageBuffer counters;
layout(r32ui, binding = 4) uniform uimage1D line;
void main ()
{
  uint t0 = imageAtomicAdd(counters, int(uvec4(gl_GlobalInvocationID.x, 0u, 0u, 0u).x), 1u);
  uint t1 = imageAtomicCompSwap(counters, int(uvec4(gl_GlobalInvocationID.x, 0u, 0u, 0u).x), uint((int(t0) + 1)), 0u);
  uint t2 = imageAtomicCompSwap(line, int(uvec4(gl_GlobalInvocationID.x, 0u, 0u, 0u).x), t0, 1u);
  results.values[int(gl_GlobalInvocationID.x)] = uint((int(t1) + int(t2)));
}
