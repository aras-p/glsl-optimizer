// threadsPerThreadgroup: 1, 1, 1
kernel void xlatMtlMain (constant xlatMtlShaderUniform& _mtl_u [[buffer(0)]]
  , constant _xlat_buf_Input_std140& sourceData [[buffer(1)]]
  , device _xlat_buf_Matrices_std430& matrices [[buffer(2)]]
  , device _xlat_buf_Output_std430& data [[buffer(3)]]
  , uint3 gl_GlobalInvocationID [[thread_position_in_grid]])
{
  float2 t0 = float2(sourceData.tiny.value);
  matrices.values[0] = float2(float(gl_GlobalInvocationID.x));
  matrices.values[1] = t0;
  data.entries[int(gl_GlobalInvocationID.x)].pair.a = float4(sourceData.entry.pair.a);
  data.entries[int(gl_GlobalInvocationID.x)].pair.b = uint4(sourceData.entry.pair.b);
  float4 t1 = float4(sourceData.entry.transform[0]);
  data.entries[int(gl_GlobalInvocationID.x)].transform[0] = t1;
  float4 t2 = float4(sourceData.entry.transform[1]);
  data.entries[int(gl_GlobalInvocationID.x)].transform[1] = t2;
  float4 t3 = float4(sourceData.entry.transform[2]);
  data.entries[int(gl_GlobalInvocationID.x)].transform[2] = t3;
  float4 t4 = float4(sourceData.entry.transform[3]);
  data.entries[int(gl_GlobalInvocationID.x)].transform[3] = t4;
  data.entries[(1 + int(gl_GlobalInvocationID.x))].pair.a = float4(data.entries[int(gl_GlobalInvocationID.x)].pair.a);
  data.entries[(1 + int(gl_GlobalInvocationID.x))].pair.b = uint4(data.entries[int(gl_GlobalInvocationID.x)].pair.b);
  data.entries[int(gl_GlobalInvocationID.x)].transform[0] = t1;
  data.entries[int(gl_GlobalInvocationID.x)].transform[1] = t2;
  data.entries[int(gl_GlobalInvocationID.x)].transform[2] = t3;
  data.entries[int(gl_GlobalInvocationID.x)].transform[3] = t4;
  matrices.mixed.m[0] = float4(half4(sourceData.mixed.m[0]));
  matrices.mixed.m[1] = float4(half4(sourceData.mixed.m[1]));
  matrices.mixed.m[2] = float4(half4(sourceData.mixed.m[2]));
  matrices.mixed.m[3] = float4(half4(sourceData.mixed.m[3]));
  matrices.mixed.v = float4(sourceData.mixed.v);
  matrices.row[0][0] = (t1.xyz * 0.5).x;
  matrices.row[1][0] = (t1.xyz * 0.5).y;
  matrices.row[2][0] = (t1.xyz * 0.5).z;
  matrices.row[0][1] = (t2.xyz * 0.5).x;
  matrices.row[1][1] = (t2.xyz * 0.5).y;
  matrices.row[2][1] = (t2.xyz * 0.5).z;
  matrices.small[0] = float2(sourceData.small[0].value);
  matrices.small[1] = float2(sourceData.small[1].value);
  matrices.wide[0] = t1.xy;
  matrices.wide[1] = t2.xy;
  matrices.wide[2] = t3.xy;
  matrices.tiny.value = t0;
  matrices.padded.a = float2(sourceData.padded.a);
  matrices.padded.b = float(sourceData.padded.b);
  matrices.reduced[0] = t1;
  matrices.reduced[1] = t2;
  matrices.reduced[2] = t3;
  matrices.reduced[3] = t4;
  half4 t5 = half4(matrices.reduced[1]);
  float2 t6 = float2(matrices.small[0]);
  float2 t7 = float2(matrices.wide[2]);
  float2 t8 = float2(matrices.tiny.value);
  float4 t9 = ((float4(data.entries[int(gl_GlobalInvocationID.x)].transform[1]) + float(data.entries[int(gl_GlobalInvocationID.x)].transform[2][3])) + float4(t6, float2(t5.xy)));
  matrices.result = (t9 + float4(t7, t8));
}
