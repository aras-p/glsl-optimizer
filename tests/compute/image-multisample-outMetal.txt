kernel void xlatMtlMain (constant xlatMtlShaderUniform& _mtl_u [[buffer(0)]]
  , texture2d_ms<uint, access::read> multisampleImage [[texture(0)]]
  , texture2d_ms_array<uint, access::read> multisampleLayers [[texture(1)]]
  , texture2d<uint, access::write> destination [[texture(2)]]
  , uint3 gl_GlobalInvocationID [[thread_position_in_grid]])
{
  uint t0 = gl_GlobalInvocationID.z;
  int t1 = (int(multisampleImage.read(uint2(int4(uint4(gl_GlobalInvocationID.xy, 0u, 0u)).xy), uint(int(t0))).x) + int(multisampleLayers.read(uint2(int4(uint4(gl_GlobalInvocationID.xy, 1u, 0u)).xy), uint(int4(uint4(gl_GlobalInvocationID.xy, 1u, 0u)).z), uint(int(t0))).x));
  int4 t2 = int4(t1, int2(multisampleImage.get_width(), multisampleImage.get_height()).x, int3(multisampleLayers.get_width(), multisampleLayers.get_height(), multisampleLayers.get_array_size()).z, (int(multisampleImage.get_num_samples()) + int(multisampleLayers.get_num_samples())));
  destination.write(uint4(t2), uint2(int4(uint4(gl_GlobalInvocationID.xy, 0u, 0u)).xy));
}
