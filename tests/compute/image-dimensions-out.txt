layout(local_size_x = 8) in;
layout(r32f, binding = 0) uniform image1D lineImage;
layout(r32f, binding = 1) uniform image1DArray lineLayers;
layout(r32f, binding = 2) uniform imageCube cubeImage;
layout(r32f, binding = 3) uniform imageCubeArray cubeLayers;
layout(r32f, binding = 4) uniform imageBuffer linearImage;
void main ()
{
  vec4 t0 = ((imageLoad(lineImage, int(uvec4(gl_GlobalInvocationID.x, 0u, 0u, 0u).x)) + imageLoad(lineLayers, ivec2(uvec4(gl_GlobalInvocationID.x, 1u, 0u, 0u).xy))) + imageLoad(cubeImage, ivec3(uvec4(gl_GlobalInvocationID.x, 0u, 5u, 0u).xyz)));
  vec4 t1 = ((t0 + imageLoad(cubeLayers, ivec3(uvec4(gl_GlobalInvocationID.x, 0u, 8u, 0u).xyz))) + imageLoad(linearImage, int(uvec4(gl_GlobalInvocationID.x, 0u, 0u, 0u).x)));
  imageStore(lineImage, int(uvec4(gl_GlobalInvocationID.x, 0u, 0u, 0u).x), (t1 + float(imageSize(lineImage))));
  vec2 t2 = vec2(imageSize(lineLayers));
  imageStore(lineLayers, ivec2(uvec4(gl_GlobalInvocationID.x, 1u, 0u, 0u).xy), (t1 + vec4(t2, 0.0, 0.0)));
  vec2 t3 = vec2(imageSize(cubeImage));
  imageStore(cubeImage, ivec3(uvec4(gl_GlobalInvocationID.x, 0u, 5u, 0u).xyz), (t1 + vec4(t3, 0.0, 0.0)));
  vec3 t4 = vec3(imageSize(cubeLayers));
  imageStore(cubeLayers, ivec3(uvec4(gl_GlobalInvocationID.x, 0u, 8u, 0u).xyz), (t1 + vec4(t4, 0.0)));
  imageStore(linearImage, int(uvec4(gl_GlobalInvocationID.x, 0u, 0u, 0u).x), (t1 + float(imageSize(linearImage))));
}
