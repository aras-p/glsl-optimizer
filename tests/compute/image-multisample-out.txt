#version 450
layout(local_size_x = 8) in;
layout(r32ui, binding = 0) readonly uniform uimage2DMS multisampleImage;
layout(r32ui, binding = 1) readonly uniform uimage2DMSArray multisampleLayers;
layout(r32ui, binding = 2) writeonly uniform uimage2D destination;
void main ()
{
  uint t0 = gl_GlobalInvocationID.z;
  int t1 = (int(imageLoad(multisampleImage, ivec2(uvec4(gl_GlobalInvocationID.xy, 0u, 0u).xy), int(t0)).x) + int(imageLoad(multisampleLayers, ivec3(uvec4(gl_GlobalInvocationID.xy, 1u, 0u).xyz), int(t0)).x));
  imageStore(destination, ivec2(uvec4(gl_GlobalInvocationID.xy, 0u, 0u).xy), uvec4(ivec4(t1, imageSize(multisampleImage).x, imageSize(multisampleLayers).z, (imageSamples(multisampleImage) + imageSamples(multisampleLayers)))));
}
