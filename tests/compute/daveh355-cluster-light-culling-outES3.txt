#version 310 es
layout(local_size_x = 128) in;
struct PointLight {
  highp vec4 position;
  highp vec4 color;
  highp float intensity;
  highp float radius;
};
struct Cluster {
  highp vec4 minPoint;
  highp vec4 maxPoint;
  highp uint count;
  highp uint lightIndices[100];
};
layout(std430, binding = 2) buffer lightSSBO {
  restrict readonly PointLight pointLight[];
};
layout(std430, binding = 1) buffer clusterSSBO {
  restrict Cluster clusters[];
};
uniform highp mat4 viewMatrix;
void main ()
{
  highp int r0;
  highp int r1;
  highp uint cluster_lightIndices[100];
  highp int t0 = pointLight.length();
  highp int t1 = ((int(gl_WorkGroupID.x) << 7u) + int(gl_LocalInvocationID.x));
  highp vec4 t2 = clusters[t1].minPoint;
  highp vec4 t3 = clusters[t1].maxPoint;
  cluster_lightIndices = clusters[t1].lightIndices;
  r1 = 0;
  for (r0 = 0; (uint(r0) < uint(t0)); r0++) {
    highp vec4 t4 = pointLight[r0].position;
    highp vec4 t5 = (viewMatrix * t4);
    highp float t6 = pointLight[r0].radius;
    highp vec3 t7 = (min(max(t5.xyz, t2.xyz), t3.xyz) - t5.xyz);
    if ((((t6 * t6) >= dot(t7, t7)) && (uint(r1) < 100u))) {
      cluster_lightIndices[r1] = uint(r0);
      r1 = (r1 + 1);
    }
  }
  clusters[t1].count = uint(r1);
  clusters[t1].lightIndices = cluster_lightIndices;
}
