#version 310 es
layout(local_size_x=32) in;
layout(std430) readonly buffer Input { uint values[]; } inputData;
layout(std430) buffer Output { uint count; uint values[]; } outputData;
uniform uint limit;
void main()
{
    uint i = gl_GlobalInvocationID.x;
    if (i >= limit || i >= uint(inputData.values.length())) return;
    uint value = inputData.values[i];
    outputData.values[i] = value;
    memoryBarrierBuffer();
    atomicAdd(outputData.count, 1u);
    atomicCompSwap(outputData.values[i], value, value + 1u);
}
