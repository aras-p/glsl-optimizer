#version 310 es
layout(local_size_x = 8, local_size_y = 4) in;
layout(rgba16f, binding = 2) readonly uniform highp image2D sourceImage;
layout(rgba8, binding = 5) writeonly uniform highp image2D destinationImage;
uniform highp sampler2D sampledImage;
void main ()
{
  if ((!(greaterThanEqual(ivec2(gl_GlobalInvocationID.xy), imageSize(destinationImage)) != bvec2(false)))) {
    imageStore(destinationImage, ivec2(uvec4(gl_GlobalInvocationID.xy, 0u, 0u).xy), (imageLoad(sourceImage, ivec2(uvec4(gl_GlobalInvocationID.xy, 0u, 0u).xy)) + texelFetch(sampledImage, ivec2(gl_GlobalInvocationID.xy), 0)));
  }
}
