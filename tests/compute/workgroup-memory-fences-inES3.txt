#version 310 es
layout(local_size_x=8) in;
shared highp uint values[8];
layout(std430) buffer Output { uint values[]; } outputData;
void main()
{
    uint i = gl_LocalInvocationIndex;
    values[i] = i;
    if ((i & 1u) != 0u) memoryBarrierShared();
    groupMemoryBarrier();
    memoryBarrier();
    barrier();
    outputData.values[gl_GlobalInvocationID.x] = values[(i + 1u) % 8u];
}
