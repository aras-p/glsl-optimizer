precision highp float;
precision highp int;
struct particle {
    vec4 pos;
    vec4 vel;
};


layout(std430, binding=0) buffer cs_ssbo { particle prt[]; };
layout(local_size_x=64, local_size_y=1, local_size_z=1) in;

uint xorshift32(uint x) {
    x ^= x<<13;
    x ^= x>>17;
    x ^= x<<5;
    return x;
}

void main() {
    uint idx = gl_GlobalInvocationID.x;
    uint x = xorshift32(0x12345678u + idx);
    uint y = xorshift32(x);
    uint z = xorshift32(y);
    prt[idx].pos = vec4(0);
    prt[idx].vel = vec4(
        (float(x & 0x7FFFu) / 32767.0) - 0.5f,
        (float(y & 0x7FFFu) / 32767.0) * 0.5f + 2.0f,
        (float(z & 0x7FFFu) / 32767.0) - 0.5f,
        0.0);
}
