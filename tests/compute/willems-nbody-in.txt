#define WORKGROUP_SIZE 256
layout(local_size_x = WORKGROUP_SIZE) in;

struct Particle {
    vec4 pos;
    vec4 vel;
};

layout(std140, binding = 0) buffer Pos {
    Particle particles[];
};

layout(std140, binding = 1) uniform UBO {
    float deltaT;
    int particleCount;
    float gravity;
    float power;
    float soften;
} ubo;

shared vec4 sharedData[WORKGROUP_SIZE];

void main()
{
    uint index = gl_GlobalInvocationID.x;
    uint count = uint(ubo.particleCount);
    bool valid = index < count;
    vec4 position = vec4(0.0);
    if (valid)
        position = particles[index].pos;

    vec3 acceleration = vec3(0.0);
    for (uint i = 0u; i < count; i += uint(WORKGROUP_SIZE)) {
        uint source = i + gl_LocalInvocationID.x;
        if (source < count)
            sharedData[gl_LocalInvocationID.x] = particles[source].pos;
        else
            sharedData[gl_LocalInvocationID.x] = vec4(0.0);

        memoryBarrierShared();
        barrier();

        if (valid) {
            for (uint j = 0u; j < uint(WORKGROUP_SIZE); ++j) {
                vec4 other = sharedData[j];
                vec3 len = other.xyz - position.xyz;
                acceleration += ubo.gravity * len * other.w / pow(dot(len, len) + ubo.soften, ubo.power);
            }
        }

        memoryBarrierShared();
        barrier();
    }

    if (valid) {
        particles[index].vel.xyz += ubo.deltaT * acceleration;
        particles[index].vel.w += 0.1 * ubo.deltaT;
        if (particles[index].vel.w > 1.0)
            particles[index].vel.w -= 1.0;
    }
}
