// threadsPerThreadgroup: 4, 1, 1
kernel void xlatMtlMain (constant xlatMtlShaderUniform& _mtl_u [[buffer(0)]]
  , texture2d<float, access::read> sourceImage [[texture(0)]]
  , texture2d<float, access::write> destinationImage [[texture(1)]]
  , texture2d<int, access::read> sourceIntegers [[texture(2)]]
  , texture2d<int, access::write> destinationIntegers [[texture(3)]]
  , uint3 gl_GlobalInvocationID [[thread_position_in_grid]])
{
  short4 t0 = short4(sourceIntegers.read(uint2(int4(uint4(gl_GlobalInvocationID.xy, 0u, 0u)).xy)));
  destinationImage.write(float4((half4(sourceImage.read(uint2(int4(uint4(gl_GlobalInvocationID.xy, 0u, 0u)).xy))) * half(0.5))), uint2(int4(uint4(gl_GlobalInvocationID.xy, 0u, 0u)).xy));
  destinationIntegers.write(int4((t0 + short(1))), uint2(int4(uint4(gl_GlobalInvocationID.xy, 0u, 0u)).xy));
}
