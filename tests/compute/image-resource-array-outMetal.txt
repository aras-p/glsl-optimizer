kernel void xlatMtlMain (constant xlatMtlShaderUniform& _mtl_u [[buffer(0)]]
  , array<texture2d<float, access::read>, 2> sources [[texture(1)]]
  , array<texture2d<float, access::write>, 2> destinations [[texture(3)]]
  , texture2d<float> extra [[texture(0)]], sampler _mtlsmp_extra [[sampler(0)]]
  , texture2d<float, access::write> combined [[texture(5)]]
  , uint3 gl_WorkGroupID [[threadgroup_position_in_grid]]
  , uint3 gl_GlobalInvocationID [[thread_position_in_grid]])
{
  uint t0 = (gl_WorkGroupID.x & 1u);
  float4 t1 = (sources[int(t0)].read(uint2(int4(uint4(gl_GlobalInvocationID.xy, 0u, 0u)).xy)) + extra.read(uint2(int2(gl_GlobalInvocationID.xy)), uint(0)));

  float2 t2 = float2(int2(sources[int(t0)].get_width(), sources[int(t0)].get_height()));
  combined.write((t1 + float4(t2, 0.0, 0.0)), uint2(int4(uint4(gl_GlobalInvocationID.xy, 0u, 0u)).xy));
}
