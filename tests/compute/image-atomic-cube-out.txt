#version 450
layout(local_size_x = 8) in;
layout(r32ui, binding = 0) uniform uimageCube counters;
layout(r32ui, binding = 1) uniform uimageCubeArray layers;
void main ()
{
  uint t0 = imageAtomicAdd(counters, ivec3(uvec4(gl_GlobalInvocationID.xy, 4u, 0u).xyz), 1u);
  imageAtomicCompSwap(layers, ivec3(uvec4(gl_GlobalInvocationID.xy, 10u, 0u).xyz), t0, 0u);
}
