#version 310 es
layout(local_size_x = 32) in;
layout(std430) buffer Output {
  highp uint count;
  highp uint values[];
} outputData;
layout(std430) buffer Input {
  readonly highp uint values[];
} inputData;
uniform highp uint limit;
void main ()
{
  if ((gl_GlobalInvocationID.x < min(limit, uint(inputData.values.length())))) {
    highp uint t0 = gl_GlobalInvocationID.x;
    highp uint t1 = inputData.values[int(t0)];
    outputData.values[int(t0)] = t1;
    memoryBarrier();
    atomicAdd(outputData.count, 1u);
    atomicCompSwap(outputData.values[int(t0)], t1, uint((int(t1) + 1)));
  }
}
