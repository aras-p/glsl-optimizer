    vec4 t2 = (t1 / 4.0);
    vec4 t3 = mix(t2, vec4(1e+02, 1e+02, 1e+02, 1.0), equal(abs(t2), vec4((1.0/0.0))));
    ivec2 t4 = (ivec2(gl_GlobalInvocationID.xy) + params.target);
    imageStore(dest_buffer, ivec4(t4, 0, 0).xy, mix(t3, vec4(1e+02, 1e+02, 1e+02, 1.0), notEqual(t3, t3)));
  }
}
