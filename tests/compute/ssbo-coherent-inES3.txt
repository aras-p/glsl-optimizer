#version 310 es
precision highp float;
precision highp int;
layout(local_size_x=8) in;

struct Entry {
    vec4 value;
    mat4 transform;
};

layout(std430, binding=0) coherent readonly buffer Input {
    Entry entries[];
} inputData;

layout(std430, binding=1) buffer Output {
    coherent uint counter;
    uint payload;
    Entry entries[8];
} outputData;

layout(std430, binding=2) buffer Ordinary {
    uint value;
    uint results[];
} ordinary;

void main()
{
    uint i = gl_GlobalInvocationID.x;
    Entry entry = inputData.entries[i];
    outputData.entries[i] = entry;
    outputData.entries[i].transform[i & 3u][1] = entry.value.x;
    uint old = atomicCompSwap(outputData.counter, i, i + 1u);
    uint ordinaryOld = atomicCompSwap(ordinary.value, i, i + 1u);
    memoryBarrierBuffer();
    ordinary.results[i] = outputData.payload + old + ordinaryOld;
}
