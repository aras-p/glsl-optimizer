#version 430
layout(local_size_x=8) in;
shared mat2 matrices[8];
layout(std430) buffer Output { vec2 values[]; } outputData;
void main()
{
    uint i = gl_LocalInvocationIndex;
    matrices[i] = mat2(float(i));
    barrier();
    outputData.values[gl_GlobalInvocationID.x] = matrices[(i + 1u) % 8u][0];
}
