#version 310 es
layout(local_size_x = 4) in;
layout(r32ui, binding = 0) writeonly uniform highp uimage2D coherentData;
layout(r32ui, binding = 1) writeonly uniform highp uimage2D ordinaryData;
void main ()
{
  imageStore(coherentData, ivec2(uvec4(gl_GlobalInvocationID.xy, 0u, 0u).xy), uvec4(1u));
  highp ivec2 t0 = (ivec2(0, 1) + ivec2(gl_GlobalInvocationID.xy));
  imageStore(coherentData, ivec4(t0, 0, 0).xy, uvec4(1u));
  imageStore(ordinaryData, ivec2(uvec4(gl_GlobalInvocationID.xy, 0u, 0u).xy), uvec4(2u));
  memoryBarrierImage();
}
