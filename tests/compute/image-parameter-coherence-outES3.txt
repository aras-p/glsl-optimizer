#version 310 es
layout(local_size_x=4) in;
layout(r32ui, binding=0) writeonly coherent uniform highp uimage2D coherentData;
layout(r32ui, binding=1) writeonly uniform highp uimage2D ordinaryData;
void main ()
{
  highp ivec2 tmpvar_1;
  tmpvar_1 = ivec2(gl_GlobalInvocationID.xy);
  imageStore (coherentData, tmpvar_1, uvec4(1u, 1u, 1u, 1u));
  imageStore (coherentData, (tmpvar_1 + ivec2(0, 1)), uvec4(1u, 1u, 1u, 1u));
  imageStore (ordinaryData, tmpvar_1, uvec4(2u, 2u, 2u, 2u));
  memoryBarrierImage ();
}
