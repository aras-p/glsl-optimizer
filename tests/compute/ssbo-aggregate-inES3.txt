#version 310 es
layout(local_size_x=4) in;
struct Child { vec3 position; float weight; bool enabled; };
struct Entry { Child children[2]; mediump vec2 weights[3]; mat2x3 transform; };
layout(std140, row_major) uniform Input { Entry entry; Entry entries[2]; vec2 weights[2][3]; } sourceData;
layout(std430) buffer Output { Entry entries[4]; Entry pair[2]; vec2 weights[2][3]; mat2x3 transform; } outputData;
layout(std140, row_major) buffer RowOutput { Entry entry; } rowData;
void main()
{
    uint i = gl_GlobalInvocationID.x;
    outputData.entries[i] = sourceData.entry;
    outputData.weights = sourceData.weights;
    outputData.pair = sourceData.entries;
    outputData.transform = sourceData.entry.transform;
    rowData.entry = sourceData.entry;
}
