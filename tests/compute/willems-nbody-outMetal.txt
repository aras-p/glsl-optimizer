// threadsPerThreadgroup: 256, 1, 1
kernel void xlatMtlMain (constant xlatMtlShaderUniform& _mtl_u [[buffer(0)]]
  , device _xlat_buf_Pos_std140& _xlat_blk_Pos [[buffer(1)]]
  , constant _xlat_buf_UBO_std140& ubo [[buffer(2)]]
  , uint3 gl_GlobalInvocationID [[thread_position_in_grid]]
  , uint3 gl_LocalInvocationID [[thread_position_in_threadgroup]])
{
  threadgroup float4 sharedData[256];
  float4 r0 = 0;
  int r1 = 0;
  int r2 = 0;
  float3 r3 = 0;
  int t0 = int(ubo.particleCount);
  bool t1 = (gl_GlobalInvocationID.x < uint(t0));
  if (t1) {
    r0 = float4(_xlat_blk_Pos.particles[int(gl_GlobalInvocationID.x)].pos);
  } else {
    r0 = float4(0.0);
  }
  r3 = float3(0.0);
  for (r2 = 0; (uint(r2) < uint(t0)); r2 += 256) {
    int t2 = (r2 + int(gl_LocalInvocationID.x));
    if ((uint(t2) < uint(t0))) {
      sharedData[int(gl_LocalInvocationID.x)] = float4(_xlat_blk_Pos.particles[t2].pos);
    } else {
      sharedData[int(gl_LocalInvocationID.x)] = float4(0.0);
    }
    threadgroup_barrier(mem_flags::mem_threadgroup);
    if (t1) {
      for (r1 = 0; (uint(r1) < 256u); r1++) {
        float4 t3 = sharedData[r1];
        float3 t4 = (t3.xyz - r0.xyz);
        r3 = (r3 + (((float(ubo.gravity) * t4) * t3.w) / pow((dot(t4, t4) + float(ubo.soften)), float(ubo.power))));
      }
    }
    threadgroup_barrier(mem_flags::mem_threadgroup);
  }
  if (t1) {
    uint t5 = gl_GlobalInvocationID.x;
    float4 t6 = float4(_xlat_blk_Pos.particles[int(t5)].vel);
    float t7 = float(ubo.deltaT);
    float3 t8 = (t6.xyz + (t7 * r3));
    float t9 = (t6.w + (0.1 * t7));
    _xlat_blk_Pos.particles[int(t5)].vel = float4(t8, t9);
    if ((1.0 < t9)) {
      _xlat_blk_Pos.particles[int(t5)].vel[3] = float4((t9 - 1.0)).w;
    }
  }
}
