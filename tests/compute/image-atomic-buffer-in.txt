#version 430
layout(local_size_x=8) in;
layout(r32ui, binding=3) uniform coherent uimageBuffer counters;
layout(std430) buffer Results { uint values[]; } results;
void main()
{
    int i = int(gl_GlobalInvocationID.x);
    uint old = imageAtomicAdd(counters, i, 1u);
    results.values[i] = imageAtomicCompSwap(counters, i, old + 1u, 0u);
}
