#version 310 es
layout(local_size_x = 4) in;
layout(rgba16f, binding = 0) readonly uniform mediump image2D sourceImage;
layout(rgba8, binding = 1) writeonly uniform mediump image2D destinationImage;
layout(rgba16i, binding = 2) readonly uniform mediump iimage2D sourceIntegers;
layout(rgba16i, binding = 3) writeonly uniform mediump iimage2D destinationIntegers;
void main ()
{
  mediump ivec4 t0 = imageLoad(sourceIntegers, ivec2(uvec4(gl_GlobalInvocationID.xy, 0u, 0u).xy));
  imageStore(destinationImage, ivec2(uvec4(gl_GlobalInvocationID.xy, 0u, 0u).xy), (imageLoad(sourceImage, ivec2(uvec4(gl_GlobalInvocationID.xy, 0u, 0u).xy)) * 0.5));
  imageStore(destinationIntegers, ivec2(uvec4(gl_GlobalInvocationID.xy, 0u, 0u).xy), (t0 + 1));
}
