#version 310 es
layout(local_size_x=4) in;
layout(rgba16f, binding=0) readonly uniform mediump image2D sourceImage;
layout(rgba8, binding=1) writeonly uniform mediump image2D destinationImage;
layout(rgba16i, binding=2) readonly uniform mediump iimage2D sourceIntegers;
layout(rgba16i, binding=3) writeonly uniform mediump iimage2D destinationIntegers;
void main ()
{
  mediump ivec4 integers_1;
  mediump vec4 color_2;
  highp ivec2 tmpvar_3;
  tmpvar_3 = ivec2(gl_GlobalInvocationID.xy);
  highp vec4 tmpvar_4;
  tmpvar_4 = imageLoad (sourceImage, tmpvar_3);
  color_2 = tmpvar_4;
  highp ivec4 tmpvar_5;
  tmpvar_5 = imageLoad (sourceIntegers, tmpvar_3);
  integers_1 = tmpvar_5;
  imageStore (destinationImage, tmpvar_3, (color_2 * 0.5));
  imageStore (destinationIntegers, tmpvar_3, (integers_1 + ivec4(1, 1, 1, 1)));
}
