  highp vec3 position;

  Child children[2];

} rowData;

  highp vec2 weights[2][3];
  layout(column_major) highp mat2x3 transform;
} outputData;
layout(std140, row_major) uniform Input {
  layout(row_major) Entry entry;
  layout(row_major) Entry entries[2];
  highp vec2 weights[2][3];
} sourceData;

{

  outputData.weights = sourceData.weights;
  outputData.pair = sourceData.entries;
  outputData.transform = sourceData.entry.transform;
  rowData.entry = sourceData.entry;
}
