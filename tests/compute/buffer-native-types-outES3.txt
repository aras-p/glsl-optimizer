#version 310 es
layout(local_size_x = 1) in;
struct Mixed {
  mediump mat4 m;
  highp vec4 v;
};
struct Tiny {
  highp vec2 value;
};
struct Padded {
  highp vec2 a;
  highp float b;
};
struct Pair {
  highp vec4 a;
  highp uvec4 b;
};
struct Entry {
  Pair pair;
  highp mat4 transform;
};
layout(std430) buffer Matrices {
  Mixed mixed;
  highp vec2 values[2];
  highp mat2 small;
  highp mat3x2 wide;
  layout(row_major) highp mat2x3 row;
  Tiny tiny;
  Padded padded;
  mediump mat4 reduced;
  highp vec4 result;
} matrices;
layout(std430) buffer Output {
  Entry entries[];
} data;
layout(std140) uniform Input {
  Entry entry;
  Mixed mixed;
  highp mat2 small;
  Tiny tiny;
  Padded padded;
} sourceData;
void main ()
{
  matrices.values[0] = vec2(float(gl_GlobalInvocationID.x));
  matrices.values[1] = sourceData.tiny.value;
  data.entries[int(gl_GlobalInvocationID.x)].pair.a = sourceData.entry.pair.a;
  data.entries[int(gl_GlobalInvocationID.x)].pair.b = sourceData.entry.pair.b;
  data.entries[int(gl_GlobalInvocationID.x)].transform[0] = sourceData.entry.transform[0];
  data.entries[int(gl_GlobalInvocationID.x)].transform[1] = sourceData.entry.transform[1];
  data.entries[int(gl_GlobalInvocationID.x)].transform[2] = sourceData.entry.transform[2];
  data.entries[int(gl_GlobalInvocationID.x)].transform[3] = sourceData.entry.transform[3];
  data.entries[(1 + int(gl_GlobalInvocationID.x))].pair.a = data.entries[int(gl_GlobalInvocationID.x)].pair.a;
  data.entries[(1 + int(gl_GlobalInvocationID.x))].pair.b = data.entries[int(gl_GlobalInvocationID.x)].pair.b;
  data.entries[int(gl_GlobalInvocationID.x)].transform[0] = sourceData.entry.transform[0];
  data.entries[int(gl_GlobalInvocationID.x)].transform[1] = sourceData.entry.transform[1];
  data.entries[int(gl_GlobalInvocationID.x)].transform[2] = sourceData.entry.transform[2];
  data.entries[int(gl_GlobalInvocationID.x)].transform[3] = sourceData.entry.transform[3];
  matrices.mixed.m[0] = sourceData.mixed.m[0];
  matrices.mixed.m[1] = sourceData.mixed.m[1];
  matrices.mixed.m[2] = sourceData.mixed.m[2];
  matrices.mixed.m[3] = sourceData.mixed.m[3];
  matrices.mixed.v = sourceData.mixed.v;
  matrices.row[0] = (sourceData.entry.transform[0].xyz * 0.5);
  matrices.row[1] = (sourceData.entry.transform[1].xyz * 0.5);
  matrices.small[0] = sourceData.small[0];
  matrices.small[1] = sourceData.small[1];
  matrices.wide[0] = sourceData.entry.transform[0].xy;
  matrices.wide[1] = sourceData.entry.transform[1].xy;
  matrices.wide[2] = sourceData.entry.transform[2].xy;
  matrices.tiny.value = sourceData.tiny.value;
  matrices.padded.a = sourceData.padded.a;
  matrices.padded.b = sourceData.padded.b;
  matrices.reduced[0] = sourceData.entry.transform[0];
  matrices.reduced[1] = sourceData.entry.transform[1];
  matrices.reduced[2] = sourceData.entry.transform[2];
  matrices.reduced[3] = sourceData.entry.transform[3];
  mediump vec4 t0 = matrices.reduced[1];
  highp vec2 t1 = matrices.small[0];
  highp vec2 t2 = matrices.wide[2];
  highp vec2 t3 = matrices.tiny.value;
  matrices.result = (((data.entries[int(gl_GlobalInvocationID.x)].transform[1] + data.entries[int(gl_GlobalInvocationID.x)].transform[2][3]) + vec4(t1, t0.xy)) + vec4(t2, t3));
}
