#version 310 es
layout(local_size_x = 8) in;
struct Entry {
  highp vec4 value;
  highp mat4 transform;
};
layout(std430, binding = 2) buffer Ordinary {
  highp uint value;
  highp uint results[];
} ordinary;
layout(std430, binding = 1) buffer Output {
  coherent highp uint counter;
  highp uint payload;
  Entry entries[8];
} outputData;
layout(std430, binding = 0) coherent buffer Input {
  coherent readonly Entry entries[];
} inputData;
void main ()
{
  highp uint t0 = gl_GlobalInvocationID.x;
  highp vec4 t1 = inputData.entries[int(t0)].value;
  highp vec4 t2 = inputData.entries[int(t0)].transform[0];
  highp vec4 t3 = inputData.entries[int(t0)].transform[1];
  highp vec4 t4 = inputData.entries[int(t0)].transform[2];
  highp vec4 t5 = inputData.entries[int(t0)].transform[3];
  outputData.entries[int(t0)].value = t1;
  outputData.entries[int(t0)].transform[0] = t2;
  outputData.entries[int(t0)].transform[1] = t3;
  outputData.entries[int(t0)].transform[2] = t4;
  outputData.entries[int(t0)].transform[3] = t5;
  outputData.entries[int(t0)].transform[int((gl_GlobalInvocationID.x & 3u))].y = vec4(0.0, t1.x, 0.0, 0.0).y;
  highp int t6 = (1 + int(gl_GlobalInvocationID.x));
  highp uint t7 = atomicCompSwap(outputData.counter, t0, uint(t6));
  highp uint t8 = atomicCompSwap(ordinary.value, t0, uint(t6));
  memoryBarrier();
  ordinary.results[int(t0)] = uint(((int(outputData.payload) + int(t7)) + int(t8)));
}
