#version 310 es
layout(local_size_x=1) in;
struct Entry { uint index; vec3 value; float weights[3]; };
layout(std430) buffer Data { Entry entries[]; } data;
void main()
{
    uint i = gl_GlobalInvocationID.x + 1u;
    data.entries[data.entries[0].index] = data.entries[i++];
    data.entries[0].weights = data.entries[i].weights;
}
