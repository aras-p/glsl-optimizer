#version 310 es
layout(local_size_x = 64) in;
struct particle {
  highp vec4 pos;
  highp vec4 vel;
};
layout(std430, binding = 0) buffer cs_ssbo {
  particle prt[];
};
layout(std140, binding = 0) uniform cs_params {
  highp float dt;
  highp int num_particles;
};
void main ()
{
  if ((gl_GlobalInvocationID.x < uint(num_particles))) {
    highp vec4 t0 = prt[int(gl_GlobalInvocationID.x)].vel;
    highp vec4 t1 = vec4(t0.x, (t0.y - dt), t0.zw);
    highp vec4 t2 = (prt[int(gl_GlobalInvocationID.x)].pos + (t1 * dt));
    bool t3 = (t2.y < -2.0);
    highp vec4 t4 = (t3 ? (t1 * vec4(0.8, -0.8, 0.8, 0.0)) : t1);
    prt[int(gl_GlobalInvocationID.x)].pos = (t3 ? vec4(t2.x, -1.8, t2.zw) : t2);
    prt[int(gl_GlobalInvocationID.x)].vel = t4;
  }
}
