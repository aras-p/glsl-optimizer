template<typename I, typename T>
inline T _xlat_image_comp_swap_cube_array(I image, uint3 p, T compare, T value) {
  if (any(p >= uint3(image.get_width(), image.get_height(), 6u * image.get_array_size()))) return T(0);
  vec<T, 4> expected = vec<T, 4>(compare);
  while (!image.atomic_compare_exchange_weak(p.xy, p.z % 6u, p.z / 6u, &expected, vec<T, 4>(value)) && expected.x == compare) {}
  return expected.x;
}
